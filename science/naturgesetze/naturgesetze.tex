\end{definition}

\begin{satz}[Todesangst-Theorem]
\label{satz:angst}

\[

\]

\end{satz}

\begin{proof}
\textbf{Monotonie in $\kappa$:} Es gilt:
\[
\frac{\partial \Phi}{\partial \kappa} = A_0 \cdot \mu \cdot \frac{-\alpha(1-\kappa)^{\alpha-1}(\kappa^\beta + \epsilon) - (1-\kappa)^\alpha \beta \kappa^{\beta-1}}{(\kappa^\beta + \epsilon)^2}.
\]
FÃ¼r $\kappa \in (0,1)$, $\alpha, \beta > 0$ und $\mu > 0$ ist der ZÃ¤hler negativ (da beide Terme subtrahiert werden und positiv sind). Daher ist $\partial \Phi / \partial \kappa < 0$: $\Phi$ ist streng monoton fallend in $\kappa$.

\textbf{Monotonie in $\mu$:} Es gilt:
\[
\frac{\partial \Phi}{\partial \mu} = A_0 \cdot \frac{(1-\kappa)^{\alpha}}{\kappa^{\beta} + \epsilon} > 0,
\]
da der ZÃ¤hler $(1-\kappa)^\alpha > 0$ fÃ¼r $\kappa < 1$ und der Nenner positiv ist. \qedhere
\end{proof}

\[
