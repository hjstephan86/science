\section{Grundlagen}

\subsection{Definitionen und Notationen}

\begin{definition}[Subgraph Isomorphismus]
	Gegeben zwei Graphen $G = (V, E)$ und $G' = (V', E')$ mit $|V| = n$ und $|V'| = m$. \end{definition}

\begin{definition}[Polynomielle Reduktion]
	Eine polynomielle Reduktion von Problem $A$ auf Problem $B$ (geschrieben $A \leq_P B$) ist eine Funktion $f: \{0,1\}^* \to \{0,1\}^*$, die in Polynomialzeit berechenbar ist, sodass fÃ¼r alle Instanzen $x$:
	\[
	x \in A \iff f(x) \in B.
	\]
\end{definition}

\begin{definition}[2-SAT]
	Eine 2-SAT-Formel ist eine Formel in Konjunktiver Normalform, in der jede Klausel hÃ¶chstens 2 Literale enthÃ¤lt. Ein 2-SAT-Instanz besteht aus Variablen $x_1, \ldots, x_n$ und Klauseln $C_1, \ldots, C_m$, wobei jede Klausel von der Form $(\ell_i \lor \ell_j)$ ist mit Literalen $\ell_i, \ell_j$.
\end{definition}

\begin{definition}[Min-Ones-2-SAT]

\end{definition}

\begin{definition}[Multicolored Clique]
	Gegeben ein Graph $G = (V, E)$ mit KnotenfÃ¤rbung $c: V \to \{1, \ldots, r\}$ und ein Parameter $k$. Das Problem fragt, ob es eine Clique der GrÃ¶ÃŸe $k$ gibt, bei der jeder Knoten eine verschiedene Farbe hat.
\end{definition}

\begin{definition}[Set-Cover]

\section{Problem 1: Min-Ones-2-SAT}

\subsection{Reduktion auf Subgraph Isomorphismus}

\begin{satz}[Reduktion Min-Ones-2-SAT $\leq_P$ Subgraph Iso]
	Gegeben eine 2-SAT-Formel $\phi$ mit $n$ Variablen und $m$ Klauseln sowie Parameter $k$, kann die Instanz in Polynomialzeit in eine Ã¤quivalente Instanz des Subgraph Isomorphismus-Problems transformiert werden.
\end{satz}

\begin{proof}
	Wir konstruieren zwei Graphen $G$ und $G'$ wie folgt:
	
	\textbf{Konstruktion von $G$ (Variable-Knoten-Graph):}
	
	FÃ¼r jede Variable $x_i$ erstelle zwei Knoten $v_i^+$ (wahr) und $v_i^-$ (falsch). Dies ergibt insgesamt $2n$ Knoten.

	\begin{itemize}

		\begin{itemize}
			\item Wenn $l_i = x_a$ (positives Literal): FÃ¼ge Kante $(v_a^+, v_{\text{clause}})$ hinzu
			\item Wenn $l_i = \neg x_a$ (negatives Literal): FÃ¼ge Kante $(v_a^-, v_{\text{clause}})$ hinzu
		\end{itemize}
		\item Erstelle fÃ¼r jede Klausel einen separaten Knoten $v_{\text{clause}}$
	\end{itemize}
	
	Der Graph $G$ hat somit $n_G = 2n + m$ Knoten.
	
	\textbf{Konstruktion von $G'$ (ErfÃ¼llungs-Graph):}
	
	Der Graph $G'$ wird konstruiert als ReprÃ¤sentation aller mÃ¶glichen Wahrheitswertbelegungen:
	\begin{itemize}
		\item Erstelle fÃ¼r jeden mÃ¶glichen Wahrheitswertvektor $w \in \{0,1\}^n$ einen Knoten $u_w$
		\item Zwei Knoten $u_w$ und $u_{w'}$ sind verbunden, wenn die Hamming-Distanz zwischen $w$ und $w'$ gleich 1 ist (sich eine Variable unterscheidet)
		\item Dieser Graph ist der $n$-dimensionale Hypercube-Graph $Q_n$

	Der Graph $G'$ hat $n_{G'} = 2^n$ Knoten.
	
	\textbf{Kodierung der ErfÃ¼llungsbedingung:}
	
	FÃ¼r jede Klausel $(l_i \lor l_j)$ fÃ¼ge zusÃ¤tzliche Kanten zu $G'$ hinzu:
	\begin{itemize}
		\item FÃ¼r einen Knoten $u_w$ (reprÃ¤sentiert durch Vektor $w$), falls die Klausel von $w$ erfÃ¼llt wird, markiere den Knoten

	\textbf{Kodierung der Min-Ones-Bedingung:}
	
	FÃ¼r den Parameter $k$ (maximale Anzahl wahrer Variablen):
	\begin{itemize}
		\item Markiere alle Knoten $u_w$, deren Hamming-Gewicht (Anzahl Einsen) grÃ¶ÃŸer als $k$ ist

	\end{itemize}
	
	\textbf{Ã„quivalenz:}
	
	$G$ ist isomorph zu einem Subgraph von $G'_{\text{final}}$ genau dann, wenn $\phi$ eine erfÃ¼llende Belegung mit hÃ¶chstens $k$ Variablen auf wahr besitzt.
	
	\textbf{Polynomialzeit:}
	
	Die Konstruktion von $G$ erfolgt in $O(m + n)$ Zeit. Die Konstruktion von $G'$ erfordert $O(2^n \cdot n)$ Zeit fÃ¼r den Hypercube plus $O(m \cdot 2^n)$ Zeit fÃ¼r die ErfÃ¼llungskodierung. \end{proof}

\subsection{Laufzeitanalyse}

\begin{satz}[Laufzeit Min-Ones-2-SAT via Subgraph Iso]
	Die obige Reduktion ermÃ¶glicht Laufzeit $2^k \cdot \text{poly}(n, m)$ fÃ¼r Min-Ones-2-SAT mit Parameter $k$.
\end{satz}

	\]

	\]

	T_{\text{Subgraph Iso}}(2^k) = \Theta((2^k)^3) = \Theta(2^{3k}).

	T_{\text{Min-Ones-2-SAT}}(n, m, k) = O(2^{k} \cdot \text{poly}(n, m)) \text{ FPT bzgl. } k.
	\]
	
	Die Formulierung ist FPT, da alle exponentiellen Terme in $k$ erfasst werden.
\end{proof}

\subsection{Reduktion auf Subgraph Isomorphismus}

\begin{satz}[Reduktion Multicolored Clique $\leq_P$ Subgraph Iso]
	Gegeben ein Graph $G = (V, E)$ mit KnotenfÃ¤rbung $c: V \to \{1, \ldots, r\}$ und Parameter $k$, kann die Instanz in Polynomialzeit in eine Ã¤quivalente Subgraph Isomorphismus-Instanz transformiert werden.

\begin{proof}

	\textbf{Konstruktion von $H$ (Multicolored Clique Pattern):}

		\end{itemize}

		\item Verbinde $\text{color}_i$ mit allen Knoten aus $V_i$ in $G'$

	\end{itemize}

		\item Nach Konstruktion von $G'$ bedeutet dies $(v_i, v_j) \in E_G$ UND $c(v_i) \neq c(v_j)$

\end{proof}

		\item Verbinde alle Mengen-Knoten untereinander: vollstÃ¤ndiger Graph auf den $m$ Mengen-Knoten
		\item Dies erzeugt: $t_i^S \leftrightarrow t_j^S$ fÃ¼r alle $i \neq j$
	\end{itemize}
	
	Der Graph $T$ hat $n_T = m + |U|$ Knoten.
	
	\textbf{Isomorphie-Bedingung:}
	
	Ein Subgraph Isomorphismus von $P$ nach $T$ bedeutet:
	\begin{itemize}
		\item Die $k$ Knoten $s_1, \ldots, s_k$ werden auf $k$ Mengen-Knoten $t_{i_1}^S, \ldots, t_{i_k}^S$ abgebildet
		\item Da alle $s_i$ untereinander verbunden sind und alle $t_i^S$ untereinander verbunden sind (dies ist erfÃ¼llt durch die vollstÃ¤ndige Verbindung)
		\item FÃ¼r jeden Element-Knoten $p_e$ muss es ein $s_j$ geben, das mit diesem verbunden ist
		\item Nach Isomorphismus-Abbildung muss $t_e^U$ mit mindestens einem der $k$ gewÃ¤hlten Mengen-Knoten verbunden sein

		\item Da dies fÃ¼r alle $|U|$ Elemente gelten muss, Ã¼berdecken die $k$ Mengen die gesamte Grundmenge $U$
	\end{itemize}
	
	\textbf{Umkehrung:}
	
	Falls eine Set-Cover mit $k$ Mengen existiert, dann existiert ein Subgraph Isomorphismus (konstruktiv nachweisbar).
	
	\textbf{Polynomialzeit:}
	
	Die Konstruktion von $P$ erfolgt in $O(|U| + k)$ Zeit. Die Konstruktion von $T$ erfolgt in $O(m \cdot |U| + m^2)$ Zeit. Insgesamt polynomial.
\end{proof}
