\begin{lemma}[Existenz des BlattlÃ¤ngen-Optimums]
\label{lem:blatt_existenz}

\end{lemma}

\begin{proof}

\[

\[

\]
Da $\gamma>2$ ist $r^{\gamma-2}$ streng monoton wachsend; fÃ¼r $\delta\in(1,2)$ wÃ¤re $r^{\delta-2}$ fallend, fÃ¼r $\delta>2$ ebenfalls wachsend -- in jedem hier relevanten Fall ($\delta\in(2,3)$, $\gamma\in(3,4)$) sind beide Terme $-\gamma(\gamma-1)\kappa_1 r^{\gamma-2}$ und $-\delta(\delta-1)\kappa_2 r^{\delta-2}$ streng monoton fallend in $r$ fÃ¼r $r>0$. \end{proof}

\begin{satz}[Optimale FamiliengrÃ¶ÃŸe]
\label{satz:familie}

\]

\end{satz}

\begin{proof}

\end{proof}

\begin{satz}[Quantifizierung des Gewinnverlusts bei suboptimaler FamiliengrÃ¶ÃŸe]
