\subsection{Eigenschaften der StrafgebÃ¼hr-Funktion}

\begin{satz}[Monotonie und KonvexitÃ¤t der StrafgebÃ¼hr]
\label{satz:mono}
Die StrafgebÃ¼hr-Funktion $\pi(q) = \alpha(1-q)^{\beta}$ mit $\alpha \in (0,1)$
und $\beta > 0$ besitzt die folgenden Eigenschaften:
\begin{enumerate}
  \item $\pi$ ist streng monoton fallend auf $[0,1]$.
  \item $\pi(1) = 0$ und $\pi(0) = \alpha$.
  \item FÃ¼r $\beta > 1$ ist $\pi$ streng konvex auf $[0,1]$.
  \item FÃ¼r $\beta < 1$ ist $\pi$ streng konkav auf $[0,1]$.
  \item FÃ¼r $\beta = 1$ ist $\pi$ linear.
\end{enumerate}
\end{satz}

\begin{proof}
Wir berechnen die ersten beiden Ableitungen von $\pi$:
\[
\pi'(q) = -\alpha \beta (1-q)^{\beta-1}.
\]
Da $\alpha > 0$, $\beta > 0$ und $(1-q)^{\beta-1} \geq 0$ fÃ¼r $q \in [0,1)$,
gilt $\pi'(q) < 0$ fÃ¼r alle $q \in [0,1)$.  Damit ist $\pi$ streng monoton fallend.
\emph{(Eigenschaft 1)}

FÃ¼r $q = 1$: $\pi(1) = \alpha(1-1)^{\beta} = 0$.
FÃ¼r $q = 0$: $\pi(0) = \alpha \cdot 1^{\beta} = \alpha$.
\emph{(Eigenschaft 2)}

Die zweite Ableitung lautet:
\[
\pi''(q) = \alpha \beta (\beta - 1)(1-q)^{\beta-2}.
\]
FÃ¼r $\beta > 1$: $\pi''(q) > 0$ auf $(0,1)$ -- $\pi$ ist streng konvex.
\emph{(Eigenschaft 3)}

FÃ¼r $\beta < 1$: $\pi''(q) < 0$ auf $(0,1)$ -- $\pi$ ist streng konkav.
\emph{(Eigenschaft 4)}

FÃ¼r $\beta = 1$: $\pi''(q) = 0$ Ã¼berall -- $\pi$ ist linear.
\emph{(Eigenschaft 5)}
\end{proof}

\begin{korollar}[Differenzierbarkeit]
Die StrafgebÃ¼hr-Funktion $\pi$ ist auf $(0,1]$ beliebig oft differenzierbar.
An der Stelle $q = 0$ ist sie genau dann differenzierbar, wenn $\beta \geq 1$.
\end{korollar}

\begin{proof}
FÃ¼r $q \in (0,1]$ sind alle Ableitungen $\pi^{(n)}(q) = \alpha \cdot
\frac{\beta!}{(\beta-n)!}(-1)^n(1-q)^{\beta-n}$ (verallgemeinert fÃ¼r reelles $\beta$)
stetig.  Bei $q = 0$ existiert der Grenzwert von $\pi'(0^+) = -\alpha\beta$
genau dann endlich, wenn $\beta - 1 \geq 0$, also $\beta \geq 1$.
\end{proof}

\subsection{Eigenschaften der Gutschrift-Funktion}

\begin{satz}[Grundeigenschaften der Gutschrift-Funktion]
\label{satz:gutt}
Die Gutschrift-Funktion $G(N,q) = N(1 - \alpha(1-q)^{\beta})$ erfÃ¼llt:

  \item $G(N, 1) = N$ -- Vollgutschrift fÃ¼r neuwertiges Bargeld.

  \item $G$ ist streng monoton steigend in $q$ fÃ¼r festes $N > 0$.

\begin{definition}[Betriebskostenfunktion]

der als Einnahme verbucht wird: $r \cdot N \cdot \pi(q)$).
\end{definition}

\begin{satz}[Existenz und Eindeutigkeit des optimalen Schwellwerts]
\label{satz:schwellwert}
FÃ¼r festes $N > 0$ und Betriebskosten $C_{\text{op}} > 0$ existiert ein
eindeutiger Schwellwert $q^*(N) \in [0,1)$, sodass:
\[

\]
FÃ¼r $q < q^*(N)$ ist eine StrafgebÃ¼hr hÃ¶her als die Betriebskosten -- die Transaktion ist
profitabel.  FÃ¼r $q > q^*(N)$ ist die StrafgebÃ¼hr geringer als die Betriebskosten.
\end{satz}

\begin{proof}

Da $h$ nach Satz~\ref{satz:mono} streng monoton fallend und stetig ist, mit
$h(0) = rN\alpha$ und $h(1) = 0$, existiert nach dem Zwischenwertsatz genau
dann ein $q^* \in (0,1)$ mit $h(q^*) = C_{\text{op}}$, wenn $C_{\text{op}} < rN\alpha$.
In diesem Fall gilt:
\[
q^* = 1 - \left(\frac{C_{\text{op}}}{r N \alpha}\right)^{1/\beta}.
\]
Eindeutigkeit folgt aus der strengen Monotonie von $h$.
Falls $C_{\text{op}} \geq rN\alpha$, so ist $q^* = 0$ (alle QualitÃ¤ten profitabel).
\end{proof}
