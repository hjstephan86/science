\begin{definition}[Kanal]
	Ein Kanal $c_i$, $i \in \{0,\ldots,n-1\}$, ist eine unabhÃ¤ngige Realisierung einer Funktion $f: X \to Y$, die fÃ¼r eine Eingabe $x \in X$ eine Ausgabe $c_i(x) \in Y$ liefert.

\begin{satz}[Echte Implikation]

	FÃ¼r jeden Mehrheits-Voter $V_n$ mit $n \geq 3$ und jede Eingabe $x$ gilt:

\end{proof}

\begin{proof}
	Folgt unmittelbar aus Satz~\ref{satz:implikation}: Im dort konstruierten Zustand ist $S(x)=1$ (Safety-Bewertung: sicher) und $T(x)=0$ (Security-Bewertung: unsicher) gleichzeitig wahr fÃ¼r denselben $x$. \qedhere

\subsection{Monotonie und Grenzverhalten}

\begin{lemma}[Monotonie in $n$]
	\label{lem:monotonie}
