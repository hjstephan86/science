\subsection{Beweis 1: Druckverlust im laminar-hydrodynamischen Grenzfall}
\label{ssec:beweis_lam}

\begin{theorem}[DruckverlustÃ¼berlegenheit im laminaren Bereich]
\label{thm:druckverlust_lam}

\begin{equation}

  = \frac{128\,\mu\,\dot{V}}{\pi\,N^2\,\dK^4}
  < \frac{128\,\mu\,\dot{V}}{\pi\,\dG^4}
  = \left.\frac{\Delta p}{L}\right|_{\mathrm{G}},
  \label{eq:lam_druckverlust_vergleich}
\end{equation}
sofern $N > 1$, d.\,h.\ sofern das groÃŸe Rohr durch mehr als ein kleines
ersetzt wird.
\end{theorem}

\begin{proof}
Das Hagen-Poiseuille-Gesetz fÃ¼r ein einzelnes Rohr lautet~\cite{White2016}:
\begin{equation}
  \frac{\Delta p}{L} = \frac{128\,\mu\,\dot{V}_{\mathrm{einzel}}}{\pi\,d^4}.
  \label{eq:hp}
\end{equation}
FÃ¼r das groÃŸe Rohr gilt $\dot{V}_{\mathrm{einzel}} = \dot{V}$, also:
\begin{equation}
  \left.\frac{\Delta p}{L}\right|_{\mathrm{G}}
    = \frac{128\,\mu\,\dot{V}}{\pi\,\dG^4}.
\end{equation}
FÃ¼r die parallele Anordnung aus $N$ kleinen Rohren teilt sich der Gesamtstrom
gleichmÃ¤ÃŸig auf: $\dot{V}_{\mathrm{einzel}} = \dot{V}/N$.

\begin{align}
  \left.\frac{\Delta p}{L}\right|_{\mathrm{par, K}}
    &= \frac{128\,\mu\,(\dot{V}/N)}{\pi\,\dK^4}
     = \frac{128\,\mu\,\dot{V}}{N \cdot \pi \cdot \dG^4/N^2}
     = \frac{128\,\mu\,\dot{V} \cdot N}{\pi\,\dG^4 \cdot N^2 / N}
     \nonumber \\
    &= \frac{128\,\mu\,\dot{V}}{\pi\,N^2\,\dK^4}
     = \frac{128\,\mu\,\dot{V}}{\pi\,N\,\dG^4}.
\end{align}
Damit:
\begin{equation}
  \frac{\left.\Delta p/L\right|_{\mathrm{par,K}}}
       {\left.\Delta p/L\right|_{\mathrm{G}}}
  = \frac{1}{N} < 1 \quad \text{fÃ¼r } N > 1.
\end{equation}
Das parallele Netz kleiner Rohre hat also einen um den Faktor $N$ geringeren
Druckverlust als das einzelne groÃŸe Rohr.
\end{proof}

\begin{corollary}
\label{cor:druckverlust_lam}
Der Druckverlustvorteil wÃ¤chst quadratisch mit dem VerhÃ¤ltnis der Durchmesser:
\begin{equation}

  = N = \left(\frac{\dG}{\dK}\right)^2.
\end{equation}
\end{corollary}

\begin{proof}

\begin{equation}

\end{equation}

\end{equation}

\begin{align}

  \left.\frac{\Delta p}{L}\right|_{\mathrm{kl}}
    &= 0{,}316\,\nu^{1/4}\,v^{7/4}\,\rho \cdot \frac{1}{2\,d_{\mathrm{K}}^{5/4}}.
\end{align}

\begin{equation}

\end{equation}

einzelnen kleinen Rohres (Parallel\-schaltung gleicher WiderstÃ¤nde).

\noindent

\noindent

$R_{\mathrm{par}} = R_{\mathrm{kl}} / N$.

\noindent

\end{equation}

\end{equation}

\subsection{Beweis 3: Spezifische WÃ¤rmeÃ¼bertragungsflÃ¤che}
\label{ssec:beweis_waerme}

\begin{theorem}[GrÃ¶ÃŸere spezifische WÃ¤rmeÃ¼bertragungsflÃ¤che kleiner Rohre]
\label{thm:waermeflaeche}
FÃ¼r eine parallele Anordnung von $N$ kleinen Rohren mit Durchmesser $\dK$

  \label{eq:waermeflaeche}
\end{equation}

\end{theorem}

\begin{proof}

  WandflÃ¤che parallele kleine Rohre: $A_{\mathrm{K}} = N \cdot \pi\,\dK\,L$.\\

      = \frac{N\,\dK}{\dG}

\end{proof}

\end{corollary}

\subsection{Beweis 4: Strukturmechanische Vorteile (Barlow-Gleichung)}

\begin{theorem}[Geringere erforderliche Wanddicke bei kleinen Rohren]
\label{thm:barlow}

\end{equation}
\end{theorem}

\begin{proof}
